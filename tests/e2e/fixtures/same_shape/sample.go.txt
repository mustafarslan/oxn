package sample

func Classify(items []int, limit int) int {
	total := 0
	for _, item := range items {
		if item > limit {
			if item > limit*2 {
				total = total + item
			} else {
				total = total + limit
			}
		}
	}
	for total > 100 {
		total = total - 10
	}
	if total < 0 {
		return 0
	}
	return total
}
