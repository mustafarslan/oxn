pub fn classify(items: &[i64], limit: i64) -> i64 {
    let mut total: i64 = 0;
    for item in items {
        if *item > limit {
            if *item > limit * 2 {
                total = total + *item;
            } else {
                total = total + limit;
            }
        }
    }
    while total > 100 {
        total = total - 10;
    }
    if total < 0 {
        return 0;
    }
    total
}
